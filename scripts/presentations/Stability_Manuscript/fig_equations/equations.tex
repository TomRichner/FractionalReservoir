% Base size 36pt; LaTeX's natural relative sizing for fractions/sub/superscripts is preserved.
% Equations aligned on the "=" sign.
\documentclass[border=10pt,preview]{standalone}
\usepackage{amsmath}
\usepackage{amssymb}
% Shrink subscript/superscript sizes relative to the 36pt base.
% Defaults would be ~25pt (script) / ~18pt (scriptscript); these are smaller.
\DeclareMathSizes{36}{36}{15}{11}
\begin{document}
\fontsize{36}{50}\selectfont
$ $% instantiate the math fonts at 36pt so \textfont2 / \scriptfont2 are valid below
% Push subscripts further below the baseline (font's subscript-shift dimensions).
\fontdimen16\textfont2=7pt   % sub1: subscript drop, no superscript present
\fontdimen17\textfont2=7pt   % sub2: subscript drop when a superscript is present
\fontdimen16\scriptfont2=5pt % same, for nested (script-level) subscripts
\fontdimen17\scriptfont2=5pt
\begin{align*}
\frac{dx_i}{dt} &= \frac{-x_i + u_i + \sum_{j=1}^{J} w_{ij}\, b_j r_{j}}{\tau_d} \\[1.2ex]
r_i &= \phi\left( x_i - a_{0_i} - c \sum_{k=1}^{K} a_{ik} \right) \\[1.2ex]
\frac{da_{ik}}{dt} &= \frac{-a_{ik} + r_i}{\tau_{k}} \\[1.2ex]
\frac{db_i}{dt} &= \frac{1-b_i}{\tau_{rec}} - \frac{b_i\, r_i}{\tau_{rel}}
\end{align*}
\end{document}
